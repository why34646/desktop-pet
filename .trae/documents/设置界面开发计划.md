## 一、需求概述
开发设置窗口，包含两个功能：更改姓名和开机自启动。

## 三、实现方案

### 1. 创建 SettingsDialog 类
在 `main.py` 中新增 `SettingsDialog` 类，继承自 `QDialog`：
- 窗口大小：400×300 像素
- 居中显示
- 包含：
  - 更改姓名按钮
  - 开机自启动勾选框
  - 关闭按钮

### 2. 实现更改姓名功能
- 点击按钮后弹出输入框
- 验证输入不能为空
- 再次确认"确认名字为【XX】吗？"
- 确认后更新 `basic.txt`
- 更新主程序中的 `self.pet_name`

### 3. 实现开机自启动功能（Windows）
- 使用 `winreg` 模块操作注册表
- 注册表路径：`HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Run`
- 勾选/取消时弹出确认对话框
- 勾选状态通过读取注册表动态显示

### 4. 修改 main.py
- 实现 `show_settings()` 方法，打开设置对话框
- 添加 `load_pet_name()` 调用，更新姓名显示
- 在设置窗口中保存姓名后，主窗口菜单文本同步更新

#### 新增 SettingsDialog 类
```python
class SettingsDialog(QDialog):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.parent_window = parent
        self.init_ui()
        
    def init_ui(self):
        # 窗口设置
        self.setWindowTitle("设置")
        self.resize(400, 300)
        self.setWindowFlags(Qt.Dialog | Qt.WindowCloseButtonHint)
        
        # 布局
        layout = QVBoxLayout()
        
        # 更改姓名按钮
        change_name_btn = QPushButton("更改姓名")
        change_name_btn.clicked.connect(self.change_name)
        layout.addWidget(change_name_btn)
        
        # 开机自启动勾选框
        self.autostart_checkbox = QCheckBox("开机自启动")
        self.autostart_checkbox.stateChanged.connect(self.toggle_autostart)
        self.load_autostart_status()
        layout.addWidget(self.autostart_checkbox)
        
        layout.addStretch()
        
        # 关闭按钮
        close_btn = QPushButton("关闭")
        close_btn.clicked.connect(self.accept)
        layout.addWidget(close_btn)
        
        self.setLayout(layout)
    
    def change_name(self):
        # 输入姓名
        while True:
            new_name, ok = QInputDialog.getText(self, "更改姓名", "请输入新名字：")
            if not ok:
                return
            new_name = new_name.strip()
            if new_name:
                break
            QMessageBox.warning(self, "警告", "名字不能为空！")
        
        # 二次确认
        reply = QMessageBox.question(
            self, "确认", 
            f"确认名字为【{new_name}】吗？",
            QMessageBox.Yes | QMessageBox.No
        )
        if reply != QMessageBox.Yes:
            return
        
        # 保存到 basic.txt
        basic_path = Path(__file__).parent.parent / "basic.txt"
        try:
            with open(basic_path, 'w', encoding='utf-8') as f:
                f.write(new_name)
            # 更新主窗口
            if self.parent_window:
                self.parent_window.pet_name = new_name
            QMessageBox.information(self, "成功", "姓名已更新！")
        except Exception as e:
            QMessageBox.critical(self, "错误", f"保存失败：{str(e)}")
    
    def load_autostart_status(self):
        # 读取注册表，检查是否已设置自启动
        try:
            key = winreg.OpenKey(winreg.HKEY_CURRENT_USER, 
                                r"Software\Microsoft\Windows\CurrentVersion\Run", 
                                0, winreg.KEY_READ)
            try:
                winreg.QueryValueEx(key, "DesktopPet")
                self.autostart_checkbox.setChecked(True)
            except WindowsError:
                self.autostart_checkbox.setChecked(False)
            winreg.CloseKey(key)
        except Exception:
            self.autostart_checkbox.setChecked(False)
    
    def toggle_autostart(self, state):
        # 二次确认
        action = "开启" if state == Qt.Checked else "关闭"
        reply = QMessageBox.question(
            self, "确认", 
            f"确认{action}开机自启动吗？",
            QMessageBox.Yes | QMessageBox.No
        )
        if reply != QMessageBox.Yes:
            self.autostart_checkbox.blockSignals(True)
            self.autostart_checkbox.setChecked(not (state == Qt.Checked))
            self.autostart_checkbox.blockSignals(False)
            return
        
        # 操作注册表
        try:
            key = winreg.OpenKey(winreg.HKEY_CURRENT_USER, 
                                r"Software\Microsoft\Windows\CurrentVersion\Run", 
                                0, winreg.KEY_SET_VALUE)
            if state == Qt.Checked:
                # 添加自启动项
                script_path = Path(__file__).parent.parent / "start.py"
                python_exe = sys.executable
                command = f'"{python_exe}" "{script_path}"'
                winreg.SetValueEx(key, "DesktopPet", 0, winreg.REG_SZ, command)
            else:
                # 删除自启动项
                try:
                    winreg.DeleteValue(key, "DesktopPet")
                except WindowsError:
                    pass
            winreg.CloseKey(key)
        except Exception as e:
            QMessageBox.critical(self, "错误", f"操作失败：{str(e)}")
```

#### 修改 DesktopPet 类中的 show_settings() 方法
```python
def show_settings(self):
    dialog = SettingsDialog(self)
    dialog.exec_()
```

## 五、运行流程

1. 用户点击设置按钮
2. 弹出 400×300 设置窗口
3. 可操作：
   - 更改姓名：输入 → 验证 → 确认 → 保存
   - 开机自启动：勾选 → 确认 → 注册表操作
4. 点击关闭按钮退出设置

## 六、注意事项
- Windows 注册表操作仅在 Windows 系统下有效
- 自启动项指向 `start.py` 而非 `main.py`
- 使用 `Path` 对象处理跨平台路径
